## 改动 1：新建 repair_prompt_panel.tscn

在 `scenes/` 下新建场景。根节点 Control，结构如下：

```
RepairPromptPanel (Control)
├── Overlay (ColorRect)                     ← 半透明遮罩
└── Panel (PanelContainer)                  ← 主体面板
    └── VBoxContainer
        ├── TitleLabel (Label)              ← 山峰名称
        ├── StatusLabel (Label)             ← 状态文字
        ├── RepairButton (Button)           ← "修复山峰（20 灵气）"
        ├── Separator (HSeparator)          ← 分隔线
        ├── UpgradeTitle (Label)            ← "选择功法"
        ├── UpgradeRow (HBoxContainer)      ← 功法按钮（最多 6 个）
        │   ├── FormBtn0 (Button)           ← 御剑诀
        │   ├── FormBtn1 (Button)           ← 金刚咒
        │   └── FormBtn2 (Button)           ← 离火诀
        └── CloseButton (Button)            ← "关闭"
```

挂载脚本 `res://scripts/ui/repair_prompt_panel.gd`。

- Panel 的位置、大小、颜色
- 各 Label 的字体、字号、颜色

### 2.4 核心函数：show_prompt(state, can_repair)

替代旧的全代码生成逻辑，改用节点显示/隐藏和文案赋值：

```gdscript
func show_prompt(state: Dictionary, can_repair: bool) -> void:
    visible = true
    var repaired := bool(state.get("repaired", false))
    var form_id := str(state.get("form_id", ""))
    var form_label := str(state.get("form_label", "未选择"))

    _title_label.text = str(state.get("display_name", "未选择山峰"))

    # 状态文字
    if not repaired:
        _status_label.text = "状态：未修复"
    elif form_id.is_empty():
        _status_label.text = "状态：已修复，待选功法"
    else:
        _status_label.text = "状态：已激活 — %s" % form_label

    # 修复按钮
    _repair_btn.visible = not repaired
    _repair_btn.disabled = not can_repair

    # 功法选择区域（修复后才显示）
    _separator.visible = repaired
    _upgrade_title.visible = repaired
    _upgrade_row.visible = repaired

    # 功法按钮状态
    var form_btns := [_form_btn_0, _form_btn_1, _form_btn_2, _form_btn_3, _form_btn_4, _form_btn_5]
    for i in range(FORM_OPTIONS.size()):
        var btn := form_btns[i]
        var opt_id := str(FORM_OPTIONS[i]["id"])
        var opt := FORM_OPTIONS[i]
        btn.visible = repaired
        btn.text = "%s\n%s · 伤害%s · %s\n%s" % [opt["label"], opt["atk_type"], opt["damage"], opt["speed"], opt["trait"]]
        if repaired and form_id.is_empty():
            btn.disabled = false
            btn.modulate = Color(1, 1, 1, 1)
        elif form_id == opt_id:
            btn.disabled = true
            btn.modulate = Color(1.0, 0.96, 0.78)
        else:
            btn.disabled = true
            btn.modulate = Color(1, 1, 1, 1)
```

### 2.5 其他方法

```gdscript
func update_prompt(state: Dictionary) -> void:
    if visible:
        show_prompt(state, not bool(state.get("repaired", false)))


func _on_repair_pressed() -> void:
    var gm := get_tree().current_scene.get_node_or_null("Managers/GameManager")
    if gm and gm.has_method("repair_selected_mountain"):
        gm.repair_selected_mountain()


func _on_form_pressed(form_id: String) -> void:
    var gm := get_tree().current_scene.get_node_or_null("Managers/GameManager")
    if gm and gm.has_method("upgrade_selected_mountain"):
        gm.upgrade_selected_mountain(form_id)


func _on_close_pressed() -> void:
    visible = false
```

## 改动 4：game_manager.gd — 开战前置条件

### 4.1 新增方法：检查是否有已激活的山峰

```gdscript
func has_any_mountain_activated() -> bool:
    if not mountain_manager:
        return false
    var form_id: String = ""
    if mountain_manager.has_method("get_selected_form_id"):
        form_id = mountain_manager.get_selected_form_id()
    return not form_id.is_empty()
```

### 4.2 _set_state 中控制开战按钮

在 `_set_state()` 中，将现有的开战按钮可见性逻辑（第 81-82 行）：

```gdscript
if battle_start_button_ui:
    battle_start_button_ui.visible = (new_state == GameState.PREPARATION)
```

改为同时检查是否已有激活山峰：

```gdscript
if battle_start_button_ui:
    var can_battle := has_any_mountain_activated()
    battle_start_button_ui.visible = (new_state == GameState.PREPARATION)
    if battle_start_button_ui.has_method("set_enabled"):
        battle_start_button_ui.set_enabled(can_battle)
```

### 4.3 山峰状态变化时刷新开战按钮

在 `_on_mountain_state_changed` 末尾增加：

## 验收对照

| # | 规划验收标准 | 实现方式 |
|---|------------|---------|
| 1 | 点击未修复山峰 → 光圈 + 面板显示 | `_select_peak` 更新光圈位置，面板 show_prompt(state, true) |
| 2 | 修复成功 → 面板切换到功法选择 | `repair_selected` 改变 repaired=true，面板刷新 |
| 3 | 灵气不足 → 修复按钮置灰 | `_repair_btn.disabled = not can_repair` |
| 4 | 选择功法 → 扣除 30 灵气，激活 | `upgrade_selected` 设置 form_id，面板显示已激活 |
| 5 | 灵气不够 30 → 功法按钮置灰 | 按钮 disabled（当前未做灵气检查，后续可补） |
| 6 | 点击另一座山峰 → 光圈移动 | `_select_peak` 更新 `_selection_ring.global_position` |
| 7 | 开战后器灵按功法攻击 | 已有逻辑 `main_peak.apply_peak_profile`，无需改动 |
| 8 | 未激活任何山峰 → 开战按钮置灰 | `has_any_mountain_activated` 检查 |
| 9 | 至少激活一座 → 开战按钮可点击 | `set_enabled(true)` |
