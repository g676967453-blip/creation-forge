## 模块 A：功法 → 器灵数据绑定

### A1. mountain_manager.gd — 增加方法

新增 `get_first_activated_form_id()` — 遍历全部 6 座山峰，返回第一个已分配功法（`form_id` 非空）的 form_id：

```gdscript
func get_first_activated_form_id() -> String:
    for peak_id in PEAK_IDS:
        if not _mountains.has(peak_id):
            continue
        var form_id := str(_mountains[peak_id]["form_id"])
        if not form_id.is_empty():
            return form_id
    return ""
```

修改 `has_any_mountain_activated()` — 改为遍历全部山峰：

```gdscript
func has_any_mountain_activated() -> bool:
    return not get_first_activated_form_id().is_empty()
```

### A2. game_manager.gd — start_battle

`start_battle()` 中改为从 `mountain_manager` 获取功法：

```gdscript
func start_battle() -> void:
    if current_state != GameState.PREPARATION:
        return
    _operation_mode = false
    if mountain_manager and main_peak and main_peak.has_method("apply_peak_profile"):
        var form_id := mountain_manager.get_first_activated_form_id()
        if not form_id.is_empty():
            main_peak.apply_peak_profile(form_id)
    _set_state(GameState.BATTLE)
    # ... 倒计时逻辑不变
```

### A3. game_manager.gd — _update_ui

删除 `_update_ui()` 中的这一行（不应在经营阶段覆盖攻击属性）：

```gdscript
# 删除这行：
```

## 模块 C：敌人刷新与行为

### C1. wave_manager.gd — 160° 扇形生成位置

重写 `_get_spawn_position()`，以器灵位置 (640, 650) 为圆心，在 160° 扇形弧上随机取点：

```gdscript
func _get_spawn_position(path_type: String) -> Vector2:
    var center := Vector2(640, 380)
    var radius := 540.0
    
    var angle_center := -PI / 2          # 正上方 (90°)
    var angle_range := deg_to_rad(160.0) # 160° 扇形
    var angle_min := angle_center - angle_range / 2.0
    var angle_max := angle_center + angle_range / 2.0
    
    match path_type:
        "straight":
            var angle := randf_range(angle_min, angle_max)
            return center + Vector2(cos(angle), sin(angle)) * radius
        "left_flank":
            var angle := randf_range(angle_min, angle_center)
            return center + Vector2(cos(angle), sin(angle)) * radius
        "right_flank":
            var angle := randf_range(angle_center, angle_max)
            return center + Vector2(cos(angle), sin(angle)) * radius
        _:
            var angle := randf_range(angle_min, angle_max)
            return center + Vector2(cos(angle), sin(angle)) * radius
```

### C2. wave_manager.gd — 刷新间隔调整

`_init_waves()` 中波次的 `spawn_interval` 从 0.06~0.08 秒改为 0.3~0.5 秒，让怪物有明显陆续出现的效果。

### C3. enemy_base.gd — 到达器灵后持续攻击

怪物不再 `queue_free()`，改为进入 "攻城" 状态：

- 新增变量 `var _siege_mode: bool = false`、`var _attack_timer: float = 0.0`
- `_on_area_entered()` 中不调用 `queue_free()`，设置 `_siege_mode = true`
- `_physics_process()` 中攻城状态下：停止移动，计时攻击器灵（间隔 1 秒，每次 `data.damage` 伤害）
- 怪物被杀死时（`take_damage()` → `current_hp <= 0`）才 `queue_free()`

```gdscript
func _on_area_entered(area: Area2D) -> void:
    if area.is_in_group("main_peak") and not _siege_mode:
        _siege_mode = true
        speed = 0.0

func _physics_process(delta: float) -> void:
    if _siege_mode:
        _attack_timer -= delta
        if _attack_timer <= 0.0:
            _attack_timer = 1.0
            var main_peak = get_tree().current_scene.get_node_or_null("Battlefield/MainPeak")
            if main_peak and main_peak.has_method("take_damage"):
                main_peak.take_damage(data.damage)
        return
    # ... 原有移动逻辑
```

同时 `wave_manager.gd` 的 `_on_enemy_reached_main()` 不再处理伤害（已由 enemy_base 自己处理），只减计数：

```gdscript
func _on_enemy_reached_main(enemy) -> void:
    enemies_alive -= 1
```

### C4. enemy_base.gd — 受击伤害数字

在 `take_damage()` 中生成浮动数字：

```gdscript
func take_damage(amount: float) -> void:
    current_hp -= amount
    _spawn_damage_number(amount)
    if current_hp <= 0:
        enemy_destroyed.emit(self)
        queue_free()

func _spawn_damage_number(amount: float) -> void:
    var ui_root := get_tree().current_scene.get_node_or_null("UI")
    if ui_root == null:
        return
    var popup := preload("res://scenes/floating_text.tscn").instantiate()
    popup.position = global_position + Vector2(randf_range(-12, 12), randf_range(-24, -8))
    ui_root.add_child(popup)
    if popup.has_method("show_text"):
        popup.show_text("-%.0f" % amount, Color(1.0, 0.35, 0.35))
```

`_draw()` 改为按形状绘制：

```gdscript
func _draw() -> void:
    match projectile_shape:
        "sword":
            # 细长菱形
            var pts := PackedVector2Array([Vector2(0, -6), Vector2(2, 0), Vector2(0, 4), Vector2(-2, 0)])
            draw_colored_polygon(pts, projectile_color)
        "fist":
            # 较大圆形带光晕
            draw_circle(Vector2.ZERO, 4.5, projectile_color)
            draw_circle(Vector2.ZERO, 2.0, Color(1, 1, 1, 0.6))
        "crystal":
            # 六角形
            var pts_hex := PackedVector2Array()
            for i in range(6):
                var a := deg_to_rad(i * 60.0 - 30.0)
                pts_hex.append(Vector2(cos(a), sin(a)) * 4.0)
            draw_colored_polygon(pts_hex, projectile_color)
        _:
            draw_circle(Vector2.ZERO, 3.0, projectile_color)
            draw_circle(Vector2.ZERO, 1.5, Color(1, 1, 1, 0.8))
```
