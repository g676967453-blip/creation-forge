# 1. 草稿（低成本快速验证方向）

  --prompt "<上方 Prompt 文本>" \
  --include-tools generate_image_nano_banana_pro \
  --json --download

  --prompt "<上方 Prompt 文本>" \

  --urls "<URL>" \
